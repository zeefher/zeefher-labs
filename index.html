PERGUNTAS DE NEGÓCIO. RESPOSTAS EM SQL.
SELECT p.nome,
       SUM(i.quantidade) AS unidades
FROM produtos p
JOIN itens i ON i.produto_id = p.id
GROUP BY p.id, p.nome
ORDER BY unidades DESC;
